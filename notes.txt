def format_scoreboard(team1, score1, spaces1a, icons1a, spaces1b, icons1b, misc1,
                      team2, score2, spaces2a, icons2ai, spaces2ai, icons2aii, spaces2b, icons2b, misc2,
                      pitcher, pitch_count):
    line1 = f"| {team1.ljust(4)}{score1:<3}{spaces1a:<18}{icons1a:<2}{spaces1b:<9}{icons1b:<2}{misc1:<8}|"
    line2 = f"| {team2.ljust(4)}{score2:<2}{spaces2a:<13}{icons2ai:<2}{spaces2ai:<2}{icons2aii:<2}{spaces2b:<6}{icons2b:<2}{misc2:<8}|"
    line3 = f"|                          ● ● ○    3-2       |"
    line4 = f"|─────────────|"
    line5 = f"| {pitcher.ljust(24)}P: {str(pitch_count).ljust(10)}|"
    line6 = f"|─────────────|"

    return "\n".join(["|─────────────|", line1, line2, line3, line4, line5, line6])

 Abstract Game State    Coded Game State    Status Code     Detailed / Common State Description
 Preview                S                   S               Scheduled (Normal upcoming game)
 Preview                P                   P               Pre-Game (Lineups announced, warming up)
 Preview                U                   U               Warmup
 Live                   I                   I               In Progress (Game is actively being played)
 Live                   M                   M               Manager Challenge / Review
 Live                   E                   E               Inning Transition (Between half-innings)
 Live                   C                   C               Delayed Start (Weather, transport, etc.)
 Live                   D                   D               Delayed (In-game delay, e.g., rain delay)
 Final                  F                   F               Final (Game over, official result)
 Final                  O                   O               Game Over (Last out made, awaiting official final status verification)
 Final                  W                   W               Walked Off (Game concluded via a walk-off hit)
 Final                  C or A              A or C          Cancelled (Game will not be made up)
 Final                  D                   D               Postponed (Game scheduled for a later date)
 Final                  T                   T               Suspended (Game paused, to be resumed later)
 Final                  R                   R               Rained Out
